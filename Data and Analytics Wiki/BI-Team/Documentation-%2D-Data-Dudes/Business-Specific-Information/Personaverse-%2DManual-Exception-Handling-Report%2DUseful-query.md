**Team data** 
  SELECT kd.cre92_usernametext, 
       kd.ast_personaid, 
       wdwf.cre92_teamshortname, 
       wdwf.cre92_recordbaseteamname, 
       wdwf.ast_servicedistrictname,
       rbt.cre92_rbsystemid,
       CASE 
           WHEN wdwf.ast_whodoyouworkforid = kd.cre92_primaryteambusinessunit THEN 'Yes'
           ELSE 'No'
       END AS IsPrimaryTeam,eal.teamid,eal.[TeamCode]
FROM [personaverse].ast_kaimahidetails AS kd
INNER JOIN [personaverse].cre92_ast_kaimahidetails_ast_whodoyouworkfor AS rel
    ON kd.ast_kaimahidetailsid = rel.ast_kaimahidetailsid
INNER JOIN [personaverse].ast_whodoyouworkfor AS wdwf
    ON wdwf.ast_whodoyouworkforid = rel.ast_whodoyouworkforid
LEFT JOIN [personaverse].cre92_recordbaseteams AS rbt
    ON wdwf.cre92_recordbaseteam = rbt.cre92_recordbaseteamsid
